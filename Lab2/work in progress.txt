//ALU outside module
module ALU_new (
  input [3:0] a,
  input [3:0] b,
  input [3:0] opt,
  input c_in,
  output [3:0] out,
  output cout
);
  
  assign out = (opt == 4'b0000)? a & b:
    (opt == 4'b0001)? a | b:
    (opt == 4'b0010) ? ~ (a & b):
    (opt == 4'b0011)? ~ (a | b):
    (opt == 4'b0100)? (a ^ b):
    (opt == 4'b0101)? ~ (a ^ b):
    (opt == 4'b0110)? ~ a:
    (opt == 4'b0111)? ~ b:
    (opt == 4'b1000)? - a: 
    (opt == 4'b1001)? - b:
    (opt == 4'b1010)? ~ ://output from another ALU:
    (opt == 4'b1011)? ~ ://output from another ALU:
    (opt == 4'b1100)? a * b:
    (opt == 4'b1101)? ~ ://output from another ALU:
    (opt == 4'b1110)? ~ ://output from another ALU:
    (opt == 4'b1111)? ~ ;//output from another ALU
    
endmodule
